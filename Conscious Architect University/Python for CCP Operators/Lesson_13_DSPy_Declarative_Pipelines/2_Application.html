async def deploy_intervention(transcript: str):

        # 1. Dispatch string to Machinist (DSPy)
        prediction = coach_module(client_transcript=transcript)
        
        # 2. Inject structural payload into QA Department (Pydantic)

            selected_trigger=prediction.selected_trigger,
            cbcs_score=prediction.cbcs_score
        )
        # 3. Serialize response
        return validated.model_dump()
        
    except ValueError as val_err:
        raise HTTPException(status_code=400, detail="QA Rejection")
    except RuntimeError as r_err:
        raise HTTPException(status_code=500, detail="Machinist Failed.")
