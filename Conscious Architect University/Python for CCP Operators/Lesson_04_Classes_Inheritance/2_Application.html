import subprocess, shlex

class BashExecutionBoundary:
    def __init__(self, command: str):
        self.command = command
        
    def _sanitize(self) -> str:
        parts = shlex.split(self.command)
        if parts[0] in ["rm", "sudo", "mv"]:
            raise PermissionError("Disallowed root command.")
        return shlex.join(parts)

    def execute(self) -> dict:
        safe_cmd = self._sanitize()
        res = subprocess.run(safe_cmd, shell=True, capture_output=True)
        return {"out": res.stdout}
The Reveal: To isolate execution context explicitly. Inside a polling asynchronous loop driven by an LLM, throwing raw subprocess.run() opens the entire runtime matrix to global variables and unbounded states. The instance wrapper creates a deterministic shell bounding box limiting LLM logic bleeds mathematically.
