The Machinist: Comprehension Speed

DSPy compiles massive LLM context windows. It needs to extract relevant insights from historic transcripts extremely fast without accumulating raw loop blocks.

class PrepareContext(dspy.Module):
    def forward(self, raw_transcripts: list[dict]):
        # Extract only the 'text' where 'importance' > 0.8
        filtered_text = [
            chunk["text"]
            for chunk in raw_transcripts
            if chunk.get("importance", 0.0) > 0.8
        ]
        return dspy.Prediction(context=filtered_text)
